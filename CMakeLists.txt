cmake_minimum_required(VERSION 4.2)
project(Albedo 
    VERSION 0.1
    DESCRIPTION "Shader viewer"
    LANGUAGES CXX
)

add_executable(albedo)

target_sources(albedo 
        PRIVATE 
        src/Main.cpp
        src/Scene.cpp
        
        
        # Headers TODO biatch
        PRIVATE 
          FILE_SET HEADERS 
          BASE_DIRS include 
             FILES
          include/Scene.hpp

 )


add_subdirectory(Rem)

target_compile_features(albedo PUBLIC cxx_std_23)

# MSVC FLags
target_compile_options(albedo PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/MP;/W4;/Ox;/WX;/permissive>)

# CLANG FLAGS
set(CLANG_FLAGS
    -Wall
    -Wextra
    -Wpedantic
    -Wshadow
    -Wold-style-cast
    -Wunused
    -Wconversion
    -Wsign-conversion
    -Wdouble-promotion
    -Werror)

target_compile_options(albedo PRIVATE $<$<CXX_COMPILER_ID:Clang>:${CLANG_FLAGS}>)
# Remove console if wanted 

# Path to assets and shader as compiler definition

target_compile_definitions(albedo PRIVATE ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_compile_definitions(albedo PRIVATE SHADERS="${CMAKE_CURRENT_SOURCE_DIR}/shaders")


# Link libraries
target_link_libraries(albedo PRIVATE rem)
